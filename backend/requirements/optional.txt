# installing the GTK3 runtime separately — install this only when you are
# actually implementing `GET /invoices/{id}/pdf` (currently a 501 stub).

reportlab>=4.2

# WebSockets for live stock levels and invoice status.

channels[daphne]>=4.1,<5
channels-redis>=4.2,<5

# Celery queue dashboard. Useful in dev, must sit behind auth in production —
# it exposes task arguments, which for a payroll task means salaries.
