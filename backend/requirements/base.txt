# ---------------------------------------------------------------------------
# Runtime dependencies — everything needed to boot the API and workers.
#
# Pinning policy: a floor and a major ceiling (`>=x.y,<z`) rather than `~=x.y.z`.
# Tight patch pins across ~40 packages guarantee an unsolvable graph within
# months, because each one carries its own Django range. The floor is what we
# actually require; the ceiling is where we expect breakage.
#
# Anything optional — PDF rendering, WebSockets, queue dashboards — lives in
# optional.txt. Those pull native toolchains (GTK, Daphne) that fail loudly on
# Windows and are not needed to run the API.
# ---------------------------------------------------------------------------

# --- Framework -------------------------------------------------------------
# Django 5.1 is a hard floor: `CheckConstraint(condition=...)` arrived in 5.1
# (5.0 spells it `check=`), and every model in this codebase uses it.
Django>=5.1,<5.3
djangorestframework>=3.15,<4
djangorestframework-simplejwt[crypto]>=5.3,<6
django-filter>=24.3
drf-spectacular>=0.27,<1            # OpenAPI schema -> generated TS client
drf-spectacular-sidecar>=2024.7.1   # serves Swagger/Redoc assets locally, no CDN
django-cors-headers>=4.4,<5

# --- Database --------------------------------------------------------------
# psycopg 3. Chosen over psycopg2 for native pipeline mode and better
# server-side cursor handling on the large ledger scans.
psycopg[binary,pool]>=3.2,<4
dj-database-url>=2.2,<4

# --- Async / scheduling ----------------------------------------------------
celery[redis]>=5.4,<6
kombu>=5.4,<6
# django-celery-beat must be >=2.7. 2.6.x declares `Django<5.1`, which makes
# the whole requirements file unresolvable against our Django floor — that
# single line is what produced `ResolutionImpossible` and left the venv empty.
django-celery-beat>=2.7,<3
django-celery-results>=2.5,<3
# redis-py 6 changed the connection-pool kwargs that Django's built-in
# RedisCache forwards; stay on 5.x until that is reworked.
redis>=5.0,<6
